**What is proved:** for every finite list of bytes, the exact (arbitrary-precision integer) semantics of the real lexnet weights assigns each byte the same class as a short, readable Lean program; the proof is kernel-checked with only the standard axioms.
**How long the reference is:** `lean/Reference.lean` is 187 lines in all (149 lines of code, the rest comments and blank lines), a plain port of the 86-line `spec.py`, with no weight, network unit or generated table anywhere in it.
**What it does not mean:** it says the network equals this particular highlighting policy (with its quirks), not that the policy is correct C lexing, and not that the compiled C program, its float64 arithmetic or its weight loader behave as the exact semantics.

## The reference program, in full (`lean/Reference.lean`)

```lean
/-
  lexnet, as a short program.  Nothing in this file mentions a weight, a network
  unit or a generated table.  It is a plain Lean port of `spec.py`.

  The highlighter assigns each input byte one of eleven classes.  It works in two
  passes.  A left-to-right pass tracks comments, strings, characters, preprocessor
  lines and the current word.  A right-to-left pass then gives a whole word the
  label decided at its last byte, repairs the first `/` of a comment opener, and
  looks ahead across blanks for the `(` that marks a function name.
-/
namespace Ref

abbrev Byte := Fin 256

/-- The eleven classes. -/
def plain : Nat := 0
def keyword : Nat := 1
def type : Nat := 2
def function : Nat := 3
def identifier : Nat := 4
def number : Nat := 5
def string : Nat := 6
def character : Nat := 7
def comment : Nat := 8
def preprocessor : Nat := 9
def punctuation : Nat := 10

/-! ### Byte predicates (ASCII only) -/
def isNewline (c : Nat) : Bool := c == 10
def isBlank (c : Nat) : Bool := c == 32 || c == 9 || c == 13 || c == 11 || c == 12
def isLetter (c : Nat) : Bool :=
  (65 ≤ c && c ≤ 90) || (97 ≤ c && c ≤ 122) || c == 95   -- A-Z a-z _
def isDigit (c : Nat) : Bool := 48 ≤ c && c ≤ 57
def isWordChar (c : Nat) : Bool := isLetter c || isDigit c

/-! ### Name lists.  Only the first 14 bytes of a word are remembered. -/
def bytes (s : String) : List Nat := s.toList.map Char.toNat

def keywords : List (List Nat) := ([
  "auto", "break", "case", "const", "continue", "default", "do", "else", "enum",
  "extern", "for", "goto", "if", "inline", "register", "restrict", "return",
  "sizeof", "static", "struct", "switch", "typedef", "union", "volatile", "while",
  "_Alignas", "_Alignof", "_Atomic", "_Generic", "_Noreturn", "_Static_assert",
  "_Thread_local", "alignas", "alignof", "constexpr", "static_assert",
  "thread_local", "typeof"] : List String).map bytes

def types : List (List Nat) := ([
  "char", "double", "float", "int", "long", "short", "signed", "unsigned", "void",
  "bool", "_Bool", "_Complex", "size_t", "ssize_t", "intptr_t", "wchar_t", "time_t",
  "off_t", "pid_t", "int8_t", "int16_t", "int32_t", "int64_t", "uint8_t",
  "uint16_t", "uint32_t", "uint64_t", "FILE", "va_list", "ptrdiff_t",
  "uintptr_t"] : List String).map bytes

def constants : List (List Nat) :=
  (["NULL", "nullptr", "true", "false", "EOF"] : List String).map bytes

/-! ### Left-to-right pass -/

/-- What the left-to-right pass remembers between bytes. -/
structure Left where
  block : Bool := false          -- inside or opening a /* comment
  line : Bool := false           -- inside or opening a // comment
  inStr : Bool := false
  inChr : Bool := false
  pp : Bool := false             -- inside a preprocessor line
  escaped : Bool := false        -- previous byte was an unescaped backslash in a literal
  slash : Bool := false          -- previous byte was a `/` that may open a comment
  star : Bool := false           -- previous byte was a `*` inside a block comment
  closed : Bool := false         -- previous byte closed a block comment
  strClosed : Bool := false
  chrClosed : Bool := false
  ppBackslash : Bool := false
  wasWord : Bool := false
  wasNumber : Bool := false
  lineStart : Bool := true
  word : List Nat := []          -- the current word, at most 14 bytes

/-- What the left-to-right pass says about one byte. -/
structure Seen where
  comment : Bool
  string : Bool
  char : Bool
  pp : Bool
  word : Bool
  number : Bool
  isKeyword : Bool
  isType : Bool
  isConstant : Bool
  opener : Bool     -- this byte is the second byte of `/*` or `//`
  slash : Bool
  blank : Bool
  newline : Bool
  paren : Bool      -- this byte is `(`

def step (s : Left) (c : Nat) : Left × Seen :=
  let nl := isNewline c
  let blank := isBlank c
  let letter := isLetter c
  let digit := isDigit c
  let wordChar := isWordChar c
  let inBlock := s.block && !s.closed
  let inLine := s.line && !nl
  let inStr := s.inStr && !s.strClosed && !nl
  let inChr := s.inChr && !s.chrClosed && !nl
  let inPP := s.pp && (!nl || s.ppBackslash)
  let openBlock := s.slash && c == 42
  let openLine := s.slash && c == 47
  let blocked := inBlock || inLine || inStr || inChr
  let continuing := s.wasWord && wordChar
  let starting := letter && !s.wasNumber
  let isWord := (continuing || starting) && !blocked
  let isNumber := (wordChar || c == 46) && (s.wasNumber || digit) &&
                  !(continuing || starting || blocked)
  let word := if isWord then ((if s.wasWord then s.word ++ [c] else [c]).take 14) else []
  let next : Left :=
    { block := inBlock || openBlock
      line := inLine || openLine
      inStr := inStr || (c == 34 && !(inBlock || inLine || inChr))
      inChr := inChr || (c == 39 && !(inBlock || inLine || inStr))
      pp := inPP || (c == 35 && s.lineStart && !blocked)
      escaped := (inStr || inChr) && c == 92 && !s.escaped
      slash := c == 47 && !(blocked || openLine)
      star := inBlock && c == 42
      closed := s.star && c == 47
      strClosed := inStr && c == 34 && !s.escaped
      chrClosed := inChr && c == 39 && !s.escaped
      ppBackslash := inPP && c == 92
      wasWord := isWord
      wasNumber := isNumber
      lineStart := nl || (s.lineStart && blank)
      word := word }
  (next,
   { comment := next.block || next.line, string := next.inStr, char := next.inChr
     pp := next.pp, word := isWord, number := isNumber
     isKeyword := keywords.contains word, isType := types.contains word
     isConstant := constants.contains word
     opener := openBlock || openLine, slash := c == 47, blank := blank
     newline := nl, paren := c == 40 })

def forwardPass : Left → List Byte → List Seen
  | _, [] => []
  | s, b :: bs => let (s', seen) := step s b.val; seen :: forwardPass s' bs

/-! ### Right-to-left pass -/

/-- What the right-to-left pass remembers about the bytes to its right. -/
structure Right where
  nextWord : Bool := false      -- the next byte is part of a word
  nextOpener : Bool := false    -- the next byte completes a comment opener
  paren : Bool := false         -- a `(` follows, possibly after blanks
  keyword : Bool := false       -- the word starting at the next byte is a keyword
  type : Bool := false
  constant : Bool := false
  function : Bool := false

def label (c : Seen) (r : Right) : Right × Nat :=
  let comment := c.comment || (c.slash && r.nextOpener)
  let keyword := c.word && (if r.nextWord then r.keyword else c.isKeyword)
  let type := c.word && (if r.nextWord then r.type else c.isType)
  let constant := c.word && (if r.nextWord then r.constant else c.isConstant)
  let function := c.word &&
    (if r.nextWord then r.function else r.paren && !(c.isKeyword || c.isType || c.isConstant))
  let answer : Nat :=
    if c.string then string
    else if c.char then character
    else if comment then Ref.comment
    else if c.pp then preprocessor
    else if keyword then Ref.keyword
    else if type then Ref.type
    else if function then Ref.function
    else if constant || c.number then number
    else if c.word then identifier
    else if c.blank || c.newline then plain
    else punctuation
  ({ nextWord := c.word, nextOpener := c.opener, paren := c.paren || (c.blank && r.paren),
     keyword := keyword, type := type, constant := constant, function := function }, answer)

/-- Walk the right-to-left pass: result is the state at the left end and the labels. -/
def backwardPass : List Seen → Right × List Nat
  | [] => ({}, [])
  | c :: cs => let (r, rest) := backwardPass cs; let (r', a) := label c r; (r', a :: rest)

/-- The whole highlighter: one class per input byte. -/
def referenceClasses (bs : List Byte) : List Nat :=
  (backwardPass (forwardPass {} bs)).2

end Ref
```

## Design quirks the reference makes visible

Each row was checked equal between `spec.py`, the exact evaluator and the C binary (`quirks.py`, `evidence/quirks.json`). Class digits: 0 plain, 1 keyword, 2 type, 3 function, 4 identifier, 5 number/constant, 6 string, 7 character, 8 comment, 9 preprocessor, A punctuation.

| Input | Classes | What it shows |
| --- | --- | --- |
| `b'_Static_assertSuffix'` | `11111111111111111111` | 14-byte prefix: a longer name that starts with _Static_assert is a keyword |
| `b'static_assertX'` | `44444444444444` | 13-byte keyword plus a letter is not |
| `b'"ab\\\ncd" x'` | `6666044666` | escaped newline does not continue a string literal |
| `b'// a\\\nb c'` | `888880404` | escaped newline does not continue a // comment |
| `b'#define X \\\n  1\ny'` | `99999999999999904` | backslash-newline does continue a preprocessor line |
| `b'/*/ x */ y'` | `8888888804` | /*/ does not close itself |
| `b'1foo_bar 1e+3'` | `55555555055A5` | number runs swallow identifier characters, but stop at a sign |
| `b'a\xc3\xa9b'` | `4AA4` | non-ASCII byte is punctuation, and splits an identifier |
| `b'f \t(x) g\n(y)'` | `300A4A040A4A` | function = word then blanks then ( ; a newline stops the lookahead |
| `b'if (x) sizeof (y)'` | `110A4A01111110A4A` | keyword before ( is a keyword, not a function |
| `b'int (x)'` | `2220A4A` | type before ( stays a type |
| `b'abcdefghijklmno'` | `444444444444444` | words are cut at 14 bytes: 15-letter identifier is plain identifier |
| `b'/* if */ "while"'` | `8888888806666666` | keyword inside a comment or string stays comment/string |
| `b'a / b * c'` | `40A040A04` | lone slash and star are punctuation |

Also visible in the source: only the first 14 bytes of a word are remembered, so a name that starts with a 14-byte keyword counts as that keyword (`_Static_assert` is the only keyword that long); bytes 128-255 use the all-zero feature row, so they are punctuation and split identifiers; a newline always ends a string, character literal or `//` comment, even after a backslash; function detection needs only a word followed by blanks and `(`, with no declaration parsing; the first `/` of a comment opener is repaired by the right-to-left pass.

## What the theorem does not cover

- **float64 execution of the C binary.** `exactClasses` uses exact integers. The shipped binary computes in double precision. Earlier evidence (`REPORT.md`: interval analysis, differential runs) says no rounding occurs on reachable states, but that is not part of this theorem except where the optional section below says so.
- **The C loader and compiler.** The weight file is parsed by Python (`exact.py`) to generate the Lean; an independent audit (`audit_literals.py`) re-read the literals, but there is no formal proof of any parser, the C weight loader, the C kernel or the compiler.
- **ISO C conformance.** The theorem fixes lexnet's own policy. It makes no claim about C lexing, parsing or typing (see the quirk table).
- **Resource limits.** The theorem is over mathematical lists of any length; a real run is bounded by memory.

The backward statement covers all Boolean forward-output patterns and backward states, the head rows included.

Method (`lean/Cell.lean`, generic, no weights): a table-driven sparse-affine evaluator, an interval evaluator with a proved soundness theorem (`cell_sound`), and three input boxes for the forward cell (W=1 with counter at least 8, where `h1 < 64^7 = 2^42` follows from the word-length invariant; W=1 with counter below 8; W=0). The tables are generated from the weight file (`gen_float.py`) and tied to the generated `forward` and `backward` by `rfl` (`fwd_bridge`, `bwd_bridge`). The kernel evaluates the interval propagation (`cert_fwd`, `cert_bwd` by `decide +kernel`).

It is **not** a proof that float64 execution equals the exact semantics: there is no IEEE-754 model here (Lean's `Float` is opaque), `lexnet_kernel.h` evaluates its own specialised expressions rather than the generic row order used in the trace, and the C loader and compiler are outside the proof. What it adds is that the one precondition of exactness that is purely numeric (every intermediate integer is within 2^53) is kernel-checked rather than only measured in Python.
